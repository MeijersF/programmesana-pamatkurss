## Apstrāde
5. rezultats = (izveletais_gads - speleteja_gads + speletaja_vecums)

## Izvade
6. print(f"Sveicināts gadu aprēķinā! Ievadi gadaskaitli, un uzzini cik tev būs gadi!")
7. print(f"tev {izveletais_gads} būs {rezultats} gadi!")
